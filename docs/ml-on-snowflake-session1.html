step1_feature_engineering.py
# 1. Initialize Feature Store
from snowflake.ml.feature_store import FeatureStore, CreationMode
fs = FeatureStore(session, database, schema,
    default_warehouse=warehouse,
    creation_mode=CreationMode.CREATE_IF_NOT_EXIST)

# 2. Register PATIENT entity
entity = Entity("PATIENT", join_keys=["PATIENT_ID"])
fs.register_entity(entity)

# 3. Build features server-side (Snowpark pushdown)
feature_df = df.with_column("SHOCK_INDEX",
    F.col("HEART_RATE") / F.col("SYSTOLIC_BP")
).with_column("PULSE_PRESSURE",
    F.col("SYSTOLIC_BP") - F.col("DIASTOLIC_BP"))

# 4. Register Feature View (auto-refreshes every 1 min)
fv = FeatureView("PATIENT_FEATURES", [entity],
    feature_df, refresh_freq="1 minute")
fs.register_feature_view(fv, version="v1", block=True)

# 5. Point-in-time retrieval
training_df = fs.retrieve_feature_values(
    spine_df=spine.sample(frac=0.8), features=[fv])

step2_train.py
# Submit multi-node distributed job via ML Jobs
from snowflake.ml.jobs import submit_directory
from snowflake.ml.registry import Registry

job = submit_directory(
    dir_path="source",
    entrypoint="train.py",
    compute_pool="ML_DEMO_COMPUTE_POOL",
    stage_name="@DB.SCHEMA.JOB_PAYLOADS",
    target_instances=3,  # sets RANK + WORLD_SIZE
)
job.wait()  # streams SPCS logs live

# Evaluate the newly registered version
registry = Registry(session)
mv = registry.get_model("PATIENT_RISK_MODEL").version(version_name)
metrics = mv.get_metrics()

# Promotion gate — must pass BEFORE deploy step
if (metrics["accuracy"] < 0.80
    or metrics["f1_macro"] < 0.75):
    raise RuntimeError("Model failed promotion gate")

# Get model version from Registry and deploy
from snowflake.ml.registry import Registry

registry = Registry(session,
    database_name=db, schema_name=schema)

# Get the latest promoted version
mv = registry.get_model("PATIENT_RISK_MODEL").default

# One call deploys an auto-scaling SPCS service
mv.create_service(
    service_name="PATIENT_RISK_SERVICE",
    compute_pool="ML_DEMO_COMPUTE_POOL",
    min_instances=1,
    max_instances=3,  # auto-scale
)

# Health check: run inference on sample rows
sample = session.table(test_table).limit(3)

step5_monitor.py
# Native model monitoring via snowflake.ml.monitoring
from snowflake.ml.monitoring import ModelMonitor

monitor = ModelMonitor.create(
    session=session,
    name="PATIENT_RISK_MONITOR",
    model_name="PATIENT_RISK_MODEL",
    model_version_name=version_name,
    source_table="HEALTHCARE.INFERENCE_LOGS_VIEW",
    baseline_table="HEALTHCARE.TEST_FEATURES",
    timestamp_column="TIMESTAMP",
    prediction_column_name="RISK_LEVEL",
    id_columns=["RECORD_ID"],
    warehouse="ML_DEMO_WH",
)

status = monitor.status
# → {"status": "ACTIVE", ...}

# Drift alert triggers retrain pipeline
monitor.setup_drift_alert(
    drift_metric="psi",
    drift_threshold=0.25,
    retrain_task="PIPELINE_FEATURE_ENG_TASK",
)

-- Wrap each step in a Python Stored Procedure
CREATE OR REPLACE PROCEDURE RUN_FEATURE_ENGINEERING()
RETURNS VARIANT LANGUAGE PYTHON
RUNTIME_VERSION = '3.11'
PACKAGES = ('snowflake-ml-python', 'scikit-learn')
IMPORTS = ('@JOB_PAYLOADS/source.zip')
AS $$ def handler(session):
    from source.pipeline.step1_feature_engineering import run
    return run(get_config(), session) $$;

-- Root task: CRON weekly schedule
CREATE OR REPLACE TASK PIPELINE_FEATURE_ENG_TASK
WAREHOUSE = ML_DEMO_WH
SCHEDULE = 'USING CRON 0 2 * * 0 America/Los_Angeles'
AS CALL RUN_FEATURE_ENGINEERING();

-- Child tasks chained with AFTER
CREATE OR REPLACE TASK PIPELINE_TRAIN_TASK
AFTER PIPELINE_FEATURE_ENG_TASK
AS CALL RUN_TRAIN();

ALTER TASK PIPELINE_FEATURE_ENG_TASK RESUME;
-- Manual: EXECUTE TASK PIPELINE_FEATURE_ENG_TASK;
Failure Behavior

If EVALUATE fails the promotion gate, DEPLOY and MONITOR are skipped. The live service continues on the last approved version — no downtime.

Task Observability

All run history, durations, and errors visible in Snowsight → Data → Tasks. Configure alerts with Slack/email on failure.

CLI Management
# Rebuild from scratch
python -m source.pipeline.dag --teardown
python -m source.pipeline.dag --build
# Trigger immediately
python -m source.pipeline.dag --run
